"""Recover a manually aligned Annet control, NOT a shorthand decoder.
Usage: python recover_annet.py SOURCE.pdf EMPTY_SCRATCH_DIR
Needs PyMuPDF. No network, repository mutation, search, or OCR.
"""
from pathlib import Path
import hashlib, json, sys
import fitz

PDF_SHA256 = '243dfc316e7b7d2fb7cd282dcb503ce50aac3d44334bf93fa03e419bb017df3c'
EXPECTED = 'For there is no work nor device nor knowledge nor wisdom in the grave whither thou goest'
# Components are one-reader hypotheses, not certified stroke decompositions.
# Each x interval is in PDF points on page 17 (1-based), origin upper left.
ROWS = [
 ('For', 'FOR_FORE', 94,114, 'plate p3: for/fore'),
 ('there', 'THEN_THERE',114,124, 'plate p3: then/there'),
 ('is','s',125,134, 'plate p3: s/is'),
 ('no','n+o',134,150, 'alphabet p3; word expansion supplied by p16'),
 ('work','w+r+k',151,173, 'alphabet p3; omitted vowel, p12'),
 ('nor','n+r',176,191, 'alphabet p3; omitted vowel, p12'),
 ('device','d+v+s',194,225, 'alphabet p3; soft c may use s, p11; vowels omitted'),
 ('nor','n+r',227,242, 'repeat of outline 6; same provisional components'),
 ('knowledge','n+l+g',244,267, 'alphabet p3; silent letters/vowels omitted, p12'),
 ('nor','n+r',270,283, 'repeat of outline 6; same provisional components'),
 ('wisdom','w+s+d+m',287,324, 'alphabet p3; vowels omitted, p12'),
 ('in','n',326,336, 'plate p3; small n distinction, p15'),
 ('the','e',338,345, 'plate p3; narrow e vs broad be, p15'),
 ('grave','g+r+v',345,361, 'alphabet p3; omitted vowels, p12'),
 ('whither','w+r',362,383, 'w = with; wither/whither/other values allowed p11; composition provisional'),
 ('thou','OU_THOU',386,397, 'plate p3: ou/thou'),
 ('goest','g+s+t',400,423, 'alphabet p3; omitted vowels, p12'),
]

def main():
    if len(sys.argv) != 3:
        raise SystemExit(__doc__)
    src, out = (Path(x).resolve() for x in sys.argv[1:])
    # Refuse every detected Git worktree, including linked worktrees.
    if any((p/'.git').exists() for p in (out, *out.parents)):
        raise SystemExit('Output must be outside a Git repository.')
    if out.exists() and any(out.iterdir()):
        raise SystemExit('Use an empty scratch output directory.')
    if hashlib.sha256(src.read_bytes()).hexdigest() != PDF_SHA256:
        raise SystemExit('Wrong PDF bytes; do not mix editions.')
    assert ' '.join(x[0] for x in ROWS) == EXPECTED
    assert len(ROWS) == 17
    assert [i+1 for i,x in enumerate(ROWS) if x[0]=='nor'] == [6,8,10]
    out.mkdir(parents=True, exist_ok=True)
    doc = fitz.open(src)
    assert len(doc) >= 17 and tuple(doc[16].rect) == (0.,0.,509.,805.)
    page = doc[16]
    records=[]
    for i,(word,components,x0,x1,source) in enumerate(ROWS,1):
        rect=fitz.Rect(x0,674,x1,707)
        name=f'outline-{i:02}.png'
        page.get_pixmap(matrix=fitz.Matrix(4,4),clip=rect).save(out/name)
        records.append(dict(index=i,expected_word=word,provisional_components=components,
          pdf_page=17,bbox_pdf_points=list(rect),source=source,outline_grade='M',
          expected_word_source='PDF page 16, printed item 20',crop=name))
    for p in [1,3,7,10,11,12,13,14,15,16,17]:
        doc[p-1].get_pixmap(matrix=fitz.Matrix(1.5,1.5)).save(out/f'page-{p:02}.png')
    page.get_pixmap(matrix=fitz.Matrix(4,4),clip=fitz.Rect(94,674,424,707)).save(out/'line20.png')
    doc[0].get_pixmap(matrix=fitz.Matrix(5,5),clip=fitz.Rect(115,532,440,605)).save(out/'title-verse.png')
    (out/'alignment.json').write_text(json.dumps(records,indent=2)+'\n')
    # Human review surface: expected text is explicitly visible, so NOT a blind control.
    cells=[]
    for r in records:
        cells.append(f'<div><b>{r["index"]}. {r["expected_word"]}</b><br>'
          f'<img src="{r["crop"]}" style="height:130px;max-width:180px"><br>'
          f'{r["provisional_components"]}<br><small>{r["source"]}; M</small></div>')
    (out/'review.html').write_text('<!doctype html><meta charset="utf-8">'
      '<title>Annet manually aligned control</title><h1>Manually aligned control: not a decoder</h1>'
      '<p>Answer-aware, one-reader, provisional. Printed answer: '+EXPECTED+'</p>'
      '<img src="line20.png" style="max-width:100%"><div style="display:grid;'
      'grid-template-columns:repeat(4,1fr);gap:24px">'+''.join(cells)+'</div>')
    (out/'manifest.json').write_text(json.dumps({
      'pdf_sha256':PDF_SHA256,'pymupdf':fitz.VersionBind,
      'kind':'answer-aware manual alignment; no recognition or decoding accuracy measured',
      'outline_count':17,'outline_grades':{'M':17},'target_tokens_deciphered':0,
      'files_sha256':{p.name:hashlib.sha256(p.read_bytes()).hexdigest()
                      for p in sorted(out.iterdir()) if p.is_file()}},indent=2)+'\n')
    print(json.dumps({'outlines':17,'provisional':17,'nor_positions':[6,8,10],
      'target_test_performed':False,'output':str(out)}))

if __name__ == '__main__':
    main()
